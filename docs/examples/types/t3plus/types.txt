============================================================ module core
module core {

  // ---------------- B1. Shape: a set OTHERS extend -> iface + dyn, no enum (S4 choosing rule)
  iface Shape is show {                       // `is show`: every impl must be showable (derived is fine) (S5)
    area(self) -> Float                       // required
    name(self) -> Str                         // required
    fn describe(self) -> Str {                // PROVIDED: has a body, never overridden, resolved statically (S3)
      return "{self.name()} {self.area():.2}"
    }
    fn intersects(self, other dyn Shape) -> Bool {   // `dyn Shape` PARAMETER = borrow, no alloc, no move (S2)
      return self.area().sqrt() + other.area().sqrt() > 5.0     // toy rule
    }
  }

  type Circle { r Float }                     // private field
  type Rect   { w Float, h Float }
  impl Circle {
    fn new(r Float) -> Circle fails Str { if r <= 0.0 { fail "bad radius {r}" }; return Circle { r: r } }
  }
  impl Rect {
    fn new(w Float, h Float) -> Rect fails Str { if w <= 0.0 or h <= 0.0 { fail "bad size {w}x{h}" }; return Rect { w: w, h: h } }
  }
  impl Shape for Circle                       // one-line conformance; methods live INSIDE the type (S8)
  impl Shape for Rect
  // (the method bodies sit in the type blocks; shown inline here for brevity)
  //   Circle: area(self) -> Float { return PI * self.r * self.r }   name(self) -> Str { return "circle" }
  //   Rect:   area(self) -> Float { return self.w * self.h }        name(self) -> Str { return "rect" }

  // ---------------- B2. Widget: closed core kinds + open plugin kinds -> enum with ONE Ext(dyn) variant (S4)
  enum Event { Click, Drag(Int), Close }
  const FOCUS = "focus moves to next"

  iface WidgetKind {                          // small required set; capabilities come back as data (S3)
    render(self, depth Int) -> Str
    event(mut self, path List of Int, ev Event) -> Str
    close(take self) -> Str
    fn focus_next(self) -> Str { return FOCUS } // PROVIDED: all kinds share it, no inheritance, no stubs
  }

  enum Widget {                               // payloads are built only inside this module (minor rule)
    Panel(children List of Widget),
    Button(label Str),
    Ext(dyn WidgetKind)                       // boxed here: a store site (S2)
  }

  // Why not `Pool of dyn Widget` / handles (CHOSEN: owned tree)? The tree is strictly owning and acyclic: parents own
  // children by value. Handles would add a pool, generation checks and a dispatcher (t2: ~+7 lines per container) and
  // nothing else. A pool earns its keep only for graphs / back-references / stable ids; none here.

  // Shared routing, public so a plugin CONTAINER reuses it (G11): Close at the last step takes the child out.
  fn route(children mut List of Widget, path List of Int, ev Event) -> Str {
    match (path, ev) {
      ([], _) => return "ignored"
      ([i], Close) if i < children.len() => {
        let c = children.take_out(i)          // owned Widget; `c.close()` is `take self` on an OWNED value
        return "consumed {c.close()}; {FOCUS}"
      }
      ([i, ..rest], _) => return match children.get_mut(i) { Some(c) => c.event(rest, ev), None => "no such widget" }
    }
  }

  impl Widget {
    fn panel(children List of Widget) -> Widget { return Panel(children) }
    fn button(label Str) -> Widget fails Str { if label == "" { fail "empty label" }; return Button(label) }

    fn render(self, depth Int) -> Str {
      match self {
        Panel(children) => { var out = "{indent(depth)}Panel\n"; for c in children { out += c.render(depth + 1) }; return out }
        Button(label)   => return "{indent(depth)}Button {label}\n"
        Ext(d)          => return d.render(depth)                    // DYNAMIC: the only one, and it says Ext
      }
    }

    fn event(mut self, path List of Int, ev Event) -> Str {
      match self {                            // `mut self`: arm bindings are PLACES into self (S6)
        Panel(children) => return route(children, path, ev)          // `children` is the place self.children
        Button(label)   => match ev { Click => return "pressed {label}; {FOCUS}", _ => return "ignored" }
        Ext(d)          => return d.event(path, ev)                  // DYNAMIC
      }
    }

    fn collapse(mut self) {                   // `self = V` replaces the value, dropping the old one (S6)
      match self {
        Panel(children) => { let n = children.len(); self = Button("panel of {n}") }
        _ => {}
      }
    }

    fn close(take self) -> Str {              // consumes the widget
      match self {
        Panel(children) => return "Panel[{children.map(fn(c) { c.close() }).join(", ")}]"
        Button(label)   => return "Button {label}"
        Ext(d)          => return d.close()                          // DYNAMIC, take self on an owned dyn
      }
    }
  }

  // ---------------- B3. Money: private field + fails constructor; Account: private fields
  type Money { cents Int }                    // private: a literal is legal only inside module core
  impl Money {
    fn new(cents Int) -> Money fails Str { if cents < 0 { fail "negative money {cents}" }; return Money { cents: cents } }
    fn cents(self) -> Int { return self.cents }
    fn plus(self, o Money) -> Money { return Money { cents: self.cents + o.cents } }
    fn minus(self, o Money) -> Money? { if o.cents > self.cents { return none }; return Money { cents: self.cents - o.cents } }
  }
  type Account { owner Str, balance Money }
  impl Account {
    fn new(owner Str, balance Money) -> Account fails Str { if owner == "" { fail "empty owner" }; return Account { owner: owner, balance: balance } }
    fn deposit(mut self, m Money) { self.balance = self.balance.plus(m) }
    fn withdraw(mut self, m Money) fails Str {
      match self.balance.minus(m) {
        Some(b) => self.balance = b
        None    => fail "insufficient funds: balance {self.balance.cents()}, need {m.cents()}"
      }
    }
  }
  // transfer is a FREE function: `Pool of Account` is not our type (see NOTES, unclear point 4).
  fn transfer(pool mut Pool of Account, a Handle of Account, b Handle of Account, m Money) fails Str { /* as t3 */ }
}

============================================================ module plugin (cannot edit core)
module plugin {
  type Hexagon { side Float }
  impl Hexagon {
    fn new(side Float) -> Hexagon fails Str { if side <= 0.0 { fail "bad side {side}" }; return Hexagon { side: side } }
    fn area(self) -> Float { return 3.0 * 3.0.sqrt() / 2.0 * self.side * self.side }
    fn name(self) -> Str { return "hexagon" }
    // (b) With bare field names the classic bug was `fn grow(mut self, side Float) { side = side * 2.0 }`: it assigned
    //     the PARAMETER and left the field unchanged, silently. Now fields are only `self.side`:
    // fn grow(mut self, side Float) { side = side * 2.0 }
    //   error: cannot assign to parameter `side` (parameters are immutable). To change the field write `self.side = ...`
    fn grow(mut self, k Float) { self.side = self.side * k }
  }
  impl Shape for Hexagon                      // in the plugin's module: legal, I = core's Shape, T = ours (S8)
  //   Bug variant: `area` misspelled `aera`. Reported at `impl Shape for Hexagon`:
  //   error: Hexagon does not satisfy Shape: missing area(self) -> Float (found `aera`; did you mean `area`?)

  type SliderSpec { pub value Int, pub max Int = 10 }       // DSL-style: every field pub, defaults allowed (S7)
  type Slider { value Int, max Int }
  impl Slider {
    fn new(s SliderSpec) -> Slider fails Str { if s.value < 0 or s.value > s.max { fail "slider out of range" }; return Slider { value: s.value, max: s.max } }
    fn render(self, depth Int) -> Str { return "{indent(depth)}Slider {self.value}/{self.max}\n" }
    fn event(mut self, path List of Int, ev Event) -> Str {
      match ev {
        Drag(d) => { let old = self.value; self.value = clamp(old + d, 0, self.max); return "value {old} -> {self.value}; {self.focus_next()}" }
        _ => return "ignored"
      }
    }
    fn close(take self) -> Str { return "Slider {self.value}/{self.max}" }
  }
  impl WidgetKind for Slider                  // focus_next is provided: no stub

  // A plugin CONTAINER (the gate plain T3 failed): holds core Widgets, reuses core's route.
  type Tabs { children List of Widget }
  impl Tabs {
    fn new(children List of Widget) -> Tabs fails Str { if children.is_empty() { fail "empty tabs" }; return Tabs { children: children } }
    fn render(self, depth Int) -> Str { var out = "{indent(depth)}Tabs\n"; for c in self.children { out += c.render(depth + 1) }; return out }
    fn event(mut self, path List of Int, ev Event) -> Str { return route(self.children, path, ev) }
    fn close(take self) -> Str { return "Tabs[{self.children.map(fn(c) { c.close() }).join(", ")}]" }
  }
  impl WidgetKind for Tabs
}

fn b1() fails Str {
  let shapes = List of dyn Shape [Circle.new(1.0)?, Hexagon.new(2.0)?, Rect.new(2.0, 3.0)?, Hexagon.new(1.0)?, Circle.new(2.0)?]
  var total = 0.0
  for s in shapes { print(s.describe()); total += s.area() }     // DYNAMIC x2: a call on a `dyn Shape` value
  print("total {total:.2}")
  eprint("{shapes[0].intersects(shapes[1])} {shapes[1].show()}") // borrowed dyn argument; derived show
  // (a) take self on a borrowed list element:
  // let s = shapes[0].close()
  //   error: `close` takes self, but `shapes[0]` is a borrowed place. Use `shapes.take_out(0).close()` or `remove_at(0)`.
}

fn b2() fails Str {
  let inner = Widget.panel([Widget.button("Cancel")?])
  var root  = Widget.panel([Widget.button("OK")?, Widget.Ext(Slider.new(SliderSpec { value: 3 })?), inner])  // max defaulted
  print("build"); print(root.render(0))
  // events as in expected.txt: root.event([0], Click) ... root.event([2], Close)   -- consumes the inner panel via take_out
  // plugin container (not in expected.txt):
  var t = Widget.panel([Widget.button("A")?, Widget.Ext(Tabs.new([Widget.button("X")?, Widget.panel([Widget.button("Y")?])])?)])
  root.collapse()
}
