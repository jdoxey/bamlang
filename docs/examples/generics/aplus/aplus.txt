// A+ benchmark for D2. ILLUSTRATIVE ONLY -- NOT A SYNTAX DECISION.
// Pseudo-syntax: no semicolons, braces for blocks, `List of T`, `Map of K to V`, `Pool of T`, `Handle of T`, `Id` (untyped handle).
// Notation for ADR-0001's rules, in trailing comments:
//   MOVE  ownership transfers; the source is dead      COPY  explicit .copy() or a value-form algorithm
//   BORROW  lasts one call/statement (or one `if x =`/`with` block)      DROP  a value is freed here
//   STALE  an `else` that exists only because a Handle or Id can be stale
// Rule gaps that remain after the a-plus.md decisions are listed in NOTES.md.
// A string literal creates a fresh owned Str; `take p: T` marks a parameter that is moved in (a-plus.md, decisions 1 and 4).

// ================================================================ Part 1
struct Employee { name: Str, dept: Str, salary: Int }

fn employees() -> List of Employee
  return List.of(                                          // each Employee MOVEs into the list
    Employee { name: "Grace", dept: "sales", salary: 60000 }, Employee { name: "Bob", dept: "eng", salary: 70000 },
    Employee { name: "Dave", dept: "ops", salary: 45000 }, Employee { name: "Carol", dept: "eng", salary: 90000 },
    Employee { name: "Heidi", dept: "sales", salary: 40000 }, Employee { name: "Alice", dept: "eng", salary: 70000 },
    Employee { name: "Erin", dept: "ops", salary: 55000 }, Employee { name: "Frank", dept: "sales", salary: 40000 })

fn part1()
  print("== Part 1: algorithms over Employee ==")
  emps = employees()
  sorted = emps.copy()                                     // COPY: deep, 8 Employees x 2 Strs
  sorted.sort_by(.dept, .salary desc, .name)               // in place; the field list is the only ordering
  print("sorted by dept, salary desc, name:")
  for e in sorted                                          // BORROW: shared, per iteration
    print("  {e.dept:-5} {e.name:-6} {e.salary}")
  lo = emps.min_index_by(.salary)                          // index form: no copy (first minimum)
  hi = emps.max_index_by(.salary)
  print("min_by salary: {emps[lo].name} {emps[lo].salary}")  // BORROW: emps[lo] is a place, one statement
  print("max_by salary: {emps[hi].name} {emps[hi].salary}")
  groups: Map of Str to List of Employee = Map.new()
  for e in emps                                            // BORROW
    if not groups.has(e.dept)
      groups.insert(e.dept.copy(), List.new())             // COPY the key; the empty list MOVEs in
    groups[e.dept].push(e.copy())                          // COPY the Employee; m[k] is a place, one statement
  depts = groups.keys()                                    // COPY: a new List of Str
  depts.sort()
  print("group_by dept:")
  for d in depts
    total = 0
    with g = groups[d]                                     // BORROW (shared); traps if d is missing
      for e in g
        total += e.salary
      print("  {d:-5} count={g.len()} total={total}")
  rich = emps.filter(fn e -> e.salary > 50000)             // COPY of each kept Employee (value form)
  names = rich.map(fn e -> e.name.copy())                  // COPY of each name; closures borrow e
  print("salary > 50000: {names}")
  // scope end: DROP names, rich, depts, groups, sorted, emps (reverse order)

// ================================================================ Part 2
// ----- the LRU logic, written ONCE, over Id (the untyped handle). No key type, no value type. -----
// The cache's storage is a Pool of Entry; Lru only orders the Ids that pool issued. Lru never touches the pool.
struct Lru { cap: Int, order: List of Id }                 // order: least recent first

fn Lru.new(cap: Int) -> Lru
  return Lru { cap: cap, order: List.new() }

fn Lru.touch(self, id: Id)                                 // Id copies implicitly
  if i = self.order.index_of(id)
    self.order.remove_at(i)
  self.order.push(id)

fn Lru.full(self) -> Bool
  return self.order.len() >= self.cap

fn Lru.pop_victim(self) -> Id                              // least recent; the caller removes its entry from the pool
  return self.order.remove_at(0)                           // MOVE out (an Id: a copy)
// ----- end of generic LRU (11 code lines, shared by every use) -----

// ----- per-use glue 1: UserKey -> Profile -----
struct UserKey { tenant: Int, id: Int }
struct Profile { name: Str, email: Str }
struct PEntry { key: UserKey, val: Profile }               // the key lives in the entry so eviction can find the Map key

struct ProfileCache { lru: Lru, slots: Map of UserKey to Handle of PEntry, pool: Pool of PEntry }

fn ProfileCache.new(cap: Int) -> ProfileCache
  return ProfileCache { lru: Lru.new(cap), slots: Map.new(), pool: Pool.new() }

fn ProfileCache.get(self, k: UserKey) -> Optional of Profile
  if h = self.slots.get(k)                                 // Handle is Copy: the binding copies, the Map borrow ends at once (S7)
    if e = self.pool.get(h)                                // BORROW of self.pool
      self.lru.touch(self.pool.id(h))                      // disjoint field from self.pool: allowed
      return some(e.val.copy())                            // COPY: getters return copies
    else                                                   // STALE
      return none
  else
    return none

fn ProfileCache.put(self, k: UserKey, take v: Profile) -> Optional of UserKey
  if h = self.slots.get(k)
    self.pool[h].val = v                                   // place write: DROPs the old Profile, MOVEs v; traps if stale
    self.lru.touch(self.pool.id(h))
    return none
  evicted = none
  if self.lru.full()
    if vh = self.pool.handle(self.lru.pop_victim())        // foreign Id would trap; only staleness takes else
      old = self.pool.remove(vh)                           // MOVEs the whole entry out
      self.slots.remove(old.key)
      evicted = some(old.key)                              // partial move; old.val DROPs at scope end
    else                                                   // STALE (unreachable here)
      trap("lru victim is not in the pool")
  h = self.pool.add(PEntry { key: k, val: v })             // UserKey is plain data: copies; v MOVEs
  self.slots.insert(k, h)
  self.lru.touch(self.pool.id(h))
  return evicted
// ----- end of glue 1 -----

// ----- per-use glue 2: Str -> List of Int -----
struct QEntry { key: Str, val: List of Int }
struct QueryCache { lru: Lru, slots: Map of Str to Handle of QEntry, pool: Pool of QEntry }

fn QueryCache.new(cap: Int) -> QueryCache
  return QueryCache { lru: Lru.new(cap), slots: Map.new(), pool: Pool.new() }

fn QueryCache.get(self, k: Str) -> Optional of List of Int
  if h = self.slots.get(k)
    if e = self.pool.get(h)
      self.lru.touch(self.pool.id(h))
      return some(e.val.copy())                            // COPY: deep list copy
    else                                                   // STALE
      return none
  else
    return none

fn QueryCache.put(self, take k: Str, take v: List of Int) -> Optional of Str
  if h = self.slots.get(k)
    self.pool[h].val = v                                   // DROPs the old list, MOVEs v
    self.lru.touch(self.pool.id(h))
    return none
  evicted = none
  if self.lru.full()
    if vh = self.pool.handle(self.lru.pop_victim())
      old = self.pool.remove(vh)
      self.slots.remove(old.key)
      evicted = some(old.key)                              // MOVE to the caller
    else                                                   // STALE (unreachable here)
      trap("lru victim is not in the pool")
  h = self.pool.add(QEntry { key: k.copy(), val: v })      // COPY: the entry owns one key ...
  self.slots.insert(k, h)                                  // ... and the Map owns the moved-in one
  self.lru.touch(self.pool.id(h))
  return evicted
// ----- end of glue 2 -----

fn part2()
  print("== Part 2: LRU cache reused for two key/value pairs ==")
  pc = ProfileCache.new(2)
  qc = QueryCache.new(3)
  // Local helpers that never escape (not passed to a `take` parameter, returned or assigned elsewhere) act as
  // nested functions (S2): each reaches `pc`/`qc` at call time, exclusivity checked per call. Both capture pc.
  put_p = fn t, i, take name, take email ->      // closure parameters need `take` too: they are moved into the Profile
    print("profiles put {t}/{i}")
    if ev = pc.put(UserKey { tenant: t, id: i }, Profile { name: name, email: email })
      print("profiles evict {ev.tenant}/{ev.id}")
  get_p = fn t, i ->
    if p = pc.get(UserKey { tenant: t, id: i })            // the returned copy is bound, DROPped at block end
      print("profiles get {t}/{i} hit {p.name} <{p.email}>")
    else
      print("profiles get {t}/{i} miss")
  put_q = fn take k, take vs ->
    print("queries put {k}")
    if ev = qc.put(k, vs)                                  // MOVE both: the helper owns its arguments
      print("queries evict {ev}")
  get_q = fn k ->
    if v = qc.get(k)
      print("queries get {k} hit {v}")
    else
      print("queries get {k} miss")
  put_p(1, 1, "Ann", "ann@example.com")
  put_p(1, 2, "Ben", "ben@example.com")
  get_p(1, 1)
  put_p(2, 1, "Cy", "cy@example.com")
  get_p(1, 2)
  put_p(1, 3, "Di", "di@example.com")
  get_p(2, 1)
  get_p(1, 1)
  put_q("a", [1])
  put_q("b", [2, 2])
  put_q("c", [3])
  get_q("a")
  put_q("d", [4])
  get_q("b")
  put_q("a", [1, 1])
  put_q("e", [5])
  get_q("c")
  get_q("a")
  // F3 (review-a.md): if put forgot `self.slots.remove(old.key)`, the Map keeps a Handle to the removed slot.
  // get takes the STALE else (a miss); a later put of that key traps on `self.pool[h].val = v`. Slots are reused
  // with the next generation, so the dangling Handle never reads the new tenant. `aplus --f3` / `--f3-put`.
  // scope end: DROP qc, pc (each Pool, Map and List they own)

// ================================================================ Part 3
struct Address { street: Str, city: Str }
struct User { id: Int, name: Str, addr: Optional of Address }
struct Reply { id: Int, addr: Optional of Address }

users: Map of Int to User = Map.of(                        // module-level: main-thread state
  (1, User { id: 1, name: "Ann", addr: some(Address { street: "10 Main St", city: "Springfield" }) }),
  (2, User { id: 2, name: "Ben", addr: none }))

fn find_user(id: Int) -> Optional of User
  if u = users.get(id)                                     // BORROW; cannot be returned
    return some(u.copy())                                  // COPY
  else
    return none                                            // not stale: absent

fn part3()
  print("== Part 3: Optional pipeline across a thread ==")
  (tx, rx) = Channel.new()                                 // Channel of Reply
  worker = spawn(fn ->                                     // escaping closure: captures rx by MOVE
    for r in rx                                            // each Reply MOVEd out; DROPped per iteration
      if a = r.addr                                        // BORROW
        print("addr for {r.id}: {a.street}, {a.city}")
      else
        print("addr for {r.id}: none")
  )
  for id in 1..=3
    addr = find_user(id).and_then(fn u -> u.addr)          // consumes the Optional; moves the field out (partial move, decision 4)
    tx.send(Reply { id: id, addr: addr })                  // MOVE
  tx.close()                                               // DROP of tx also closes
  worker.join()

// ================================================================ Part 4 (A+ only): outside the stdlib
// 4a: an event bus. The user sum type is not templated.
enum Event { Joined(name: Str), Left(name: Str), Message(from: Str, text: Str) }

struct Stats { counts: Pool of Int }                       // context handed to every subscriber

// Subscribers are STORED: `subscribe` has a `take` parameter, so a closure passed to it ESCAPES (S3) and captures
// by MOVE or COPY, never by borrow. A Copy variable it mutates counts as MOVED into it (S4). State shared with the
// program therefore lives in a Pool that publish lends as an argument; the closure holds a Handle (a Copy).
struct Bus { subs: List of fn(Event, Stats) }              // `take` is part of the function type of subscribe

fn Bus.new() -> Bus
  return Bus { subs: List.new() }

fn Bus.subscribe(self, take f: fn(Event, Stats))
  self.subs.push(f)                                        // MOVE: the closure (and its captures) now live in the Bus

fn Bus.publish(self, take ev: Event, stats: Stats)         // `stats` is borrowed mutably for the call
  for s in self.subs                                       // BORROW (mutable: a closure may update its own state)
    s(ev, stats)                                           // ev is lent for the call
  // end: DROP ev

fn part4_bus()
  print("-- event bus --")
  stats = Stats { counts: Pool.new() }
  counter = stats.counts.add(0)                            // Handle of Int
  bus = Bus.new()
  bus.subscribe(fn ev, st ->                               // no captures
    match ev
      Joined(n) -> print("log: {n} joined")
      Left(n) -> print("log: {n} left")
      Message(f, t) -> print("log: {f} says {t}"))
  bus.subscribe(fn ev, st ->                               // captures `counter` (a Handle) by COPY
    match ev
      Message(_, _) -> st.counts[counter] += 1             // place write through the borrowed Stats; traps if stale
      Joined(_) -> ()
      Left(_) -> ())
  prefix = "welcome"
  seen = 0
  bus.subscribe(fn ev, st ->                               // owns its state: `prefix` and `seen` are MOVEd in
    match ev
      Joined(n) ->
        seen += 1                                          // mutated captured Copy => `seen` is moved (S4)
        print("{prefix} #{seen} {n}")
      Left(_) -> ()
      Message(_, _) -> ())
  // `prefix` and `seen` are dead here. Reading `seen` is a use-after-move error (hint: "an escaping closure
  // mutates `seen`, so it owns it; share it through a Pool and a Handle").
  bus.publish(Event.Joined("ann"), stats)
  bus.publish(Event.Joined("ben"), stats)
  bus.publish(Event.Message("ann", "hi"), stats)
  bus.publish(Event.Message("ben", "hello"), stats)
  bus.publish(Event.Left("ann"), stats)
  bus.publish(Event.Message("ben", "anyone?"), stats)
  print("messages seen: {stats.counts[counter]}")         // BORROW, one statement
  // What does NOT compile (S3, S4):
  //   n = 0
  //   bus.subscribe(fn ev, st -> n += 1)                  // escapes; mutated Copy => n moved into the closure
  //   print(n)                                            // error: use of moved `n`
  // What WOULD work: emps.sort_by_fn(fn a, b -> { n += 1; a.salary < b.salary })  -- sort_by_fn's parameter is not
  // `take`, so the closure is non-escaping and borrows n mutably for the call.
  // scope end: DROP bus (and each closure's captures), stats

// 4b: mixed widget kinds. The closed set of kinds is a sum type of typed handles; no Id is needed.
struct Button { label: Str, clicks: Int }
struct Slider { value: Int, max: Int }
struct LabelW { text: Str }

enum Widget { Button(Handle of Button), Slider(Handle of Slider), Label(Handle of LabelW) }   // plain data; copies

struct Ui { buttons: Pool of Button, sliders: Pool of Slider, labels: Pool of LabelW }

fn Ui.add_button(self, items: List of Widget, label: Str) -> Handle of Button   // `items` borrowed mutably
  h = self.buttons.add(Button { label: label.copy(), clicks: 0 })   // COPY: label is borrowed
  items.push(Widget.Button(h))
  return h
fn Ui.add_slider(self, items: List of Widget, value: Int, max: Int) -> Handle of Slider
  h = self.sliders.add(Slider { value: value, max: max })
  items.push(Widget.Slider(h))
  return h
fn Ui.add_label(self, items: List of Widget, text: Str) -> Handle of LabelW
  h = self.labels.add(LabelW { text: text.copy() })        // COPY
  items.push(Widget.Label(h))
  return h

fn Ui.name(self, w: Widget) -> Str
  match w
    Button(_) -> return "button"
    Slider(_) -> return "slider"
    Label(_) -> return "label"

fn Ui.render(self, w: Widget) -> Str                       // exhaustive match: a new kind is a compile error here
  match w
    Button(h) -> return "[{self.buttons[h].label}] clicks={self.buttons[h].clicks}"   // BORROW per use; traps if stale
    Slider(h) ->
      with s = self.sliders[h]
        return "[{"#".repeat(s.value)}{".".repeat(s.max - s.value)}] {s.value}/{s.max}"
    Label(h) -> return self.labels[h].text.copy()          // COPY: render returns an owned Str

fn part4_widgets()
  print("-- widgets --")
  ui = Ui { buttons: Pool.new(), sliders: Pool.new(), labels: Pool.new() }
  items: List of Widget = List.new()
  ok = ui.add_button(items, "OK")
  s0 = ui.add_slider(items, 0, 8)
  ui.add_label(items, "Hello")
  ui.add_button(items, "Cancel")
  vol = ui.add_slider(items, 4, 8)
  ui.buttons[ok].clicks += 1
  ui.buttons[ok].clicks += 1
  ui.sliders[vol].value = 8
  ui.sliders[s0].value = 4
  for w in items                                           // BORROW
    print("{ui.name(w)}: {ui.render(w)}")                  // the returned Str is a temporary: DROPped at statement end
  // scope end: DROP items, ui
  // Dropped variant: a record of functions (`struct WidgetKind { name, render: fn(Id) -> Str }`) cannot be used
  // here. A stored `fn(Id)` is an escaping closure and cannot borrow `ui`, so the pools would have to be module
  // state and the `Id` would carry no type. For a CLOSED set of kinds the sum type is strictly shorter and checked
  // for exhaustiveness. A record of functions only earns its place for an OPEN set of kinds (plugins), where each
  // function owns its state; that is the 4a shape, not a second widget demo.

fn part4()
  print("== Part 4 (A+ only): outside the stdlib ==")
  part4_bus()
  part4_widgets()

fn main()
  part1()
  part2()
  part3()
  part4()
